// Total length of a shift in hours (handles overnight shifts like Night 21:00–05:00)
function shiftDurationHours(slot){
  const [sh,sm] = slot.start.split(':').map(Number);
  const [eh,em] = slot.end.split(':').map(Number);
  let startMin = sh*60+sm, endMin = eh*60+em;
  if(endMin <= startMin) endMin += 24*60; // crosses midnight
  return (endMin-startMin)/60;
}
// Adds N hours to a "HH:MM" time string, wrapping past midnight
function addHoursToTime(startStr, hours){
  const [sh,sm] = startStr.split(':').map(Number);
  let totalMin = Math.round(sh*60+sm + hours*60);
  totalMin = ((totalMin % 1440) + 1440) % 1440;
  const hh = String(Math.floor(totalMin/60)).padStart(2,'0');
  const mm = String(totalMin%60).padStart(2,'0');
  return `${hh}:${mm}`;
}
// Sensible default hours for a half-day booking on a given shift (half the shift, min 1h)
function defaultHalfDayHours(slot){
  return Math.max(1, Math.round((shiftDurationHours(slot)/2)*2)/2);
}
// Real Date object for when a booking's shift starts/ends on its date (handles overnight shifts)
function shiftDateTime(dateStr, timeStr, isEnd, startTimeStr){
  const [h,m] = timeStr.split(':').map(Number);
  const dt = new Date(dateStr+'T00:00:00'); dt.setHours(h,m,0,0);
  if(isEnd && startTimeStr){
    const [sh,sm] = startTimeStr.split(':').map(Number);
    if(h*60+m <= sh*60+sm) dt.setDate(dt.getDate()+1); // end time is on the next calendar day
  }
  return dt;
}

function logNotification(msg,type='info',icon=''){
  if(!currentRole || !notifications[currentRole]) return;
  const icons={good:'✅',warn:'⚠️',bad:'⛔',info:'ℹ️'};
  notifications[currentRole].unshift({
    id:notifSeq++, type, icon:icon||icons[type], msg, time:'Just now', read:false
  });
  if(notifications[currentRole].length>30) notifications[currentRole].length = 30;
  renderNotifBell();
}
// Push a notification into an arbitrary role's inbox (used to notify meeting attendees
// who may not be the persona currently logged in).
function pushNotifForRole(role, msg, type='info', icon=''){
  if(!role || !notifications[role]) return;
  const icons={good:'✅',warn:'⚠️',bad:'⛔',info:'ℹ️'};
  notifications[role].unshift({
    id:notifSeq++, type, icon:icon||icons[type], msg, time:'Just now', read:false
  });
  if(notifications[role].length>30) notifications[role].length = 30;
  if(role===currentRole) renderNotifBell();
}
function unreadNotifCount(){
  const list = notifications[currentRole]||[];
  return list.filter(n=>!n.read).length;
}
function renderNotifBell(){
  const badge = document.getElementById('notifBadge');
  if(!badge) return;
  const count = unreadNotifCount();
  if(count>0){ badge.style.display='block'; badge.textContent = count>9 ? '9+' : String(count); }
  else{ badge.style.display='none'; }
  const roleLabel = document.getElementById('notifRoleLabel');
  if(roleLabel && personas[currentRole]) roleLabel.textContent = personas[currentRole].roleLabel;
  renderNotifList();
}
function renderNotifList(){
  const listEl = document.getElementById('notifList');
  if(!listEl) return;
  const list = notifications[currentRole]||[];
  if(!list.length){
    listEl.innerHTML = `<div class="notif-empty">No notifications yet.</div>`;
    return;
  }
  listEl.innerHTML = list.map(n=>`
    <div class="notif-item ${n.read?'':'unread'}" onclick="markNotifRead(${n.id})">
      <span class="ni-icon">${n.icon}</span>

        <div class="ni-msg">${n.msg}</div>
        <div class="ni-time">${n.time}</div>
      </div>
      ${n.read?'':'<span class="ni-dot"></span>'}
    </div>
  `).join('');
}

function timeToMins(value){
  const [hh,mm] = value.split(':').map(Number);
  return hh*60 + mm;
}
function minsToTime(total){
  const hh = Math.floor(total/60);
  const mm = total%60;
  return `${String(hh).padStart(2,'0')}:${String(mm).padStart(2,'0')}`;
}
function formatClock(value){
  const [hh,mm] = value.split(':').map(Number);
  const d = new Date(2026,6,27,hh,mm,0,0);
  return d.toLocaleTimeString('en-IN',{hour:'2-digit',minute:'2-digit',hour12:true});
}
function meetingRoomEndTime(startTime, duration){
  return minsToTime(timeToMins(startTime) + duration);
}
function roomBookingRangeLabel(startTime, endTime){
  return `${formatClock(startTime)} - ${formatClock(endTime)}`;
}
function cabinBookingOverlaps(cabinId, date, startTime, endTime, ignoreId=null){
  const start = timeToMins(startTime);
  const end = timeToMins(endTime);
  return cabinRoomBookings.some(b=>{
    if(b.cabinId!==cabinId || b.date!==date || (ignoreId && b.id===ignoreId)) return false;
    if(b.status==='rejected') return false; // rejected requests free up the slot
    const bookedStart = timeToMins(b.startTime);
    const bookedEnd = timeToMins(b.endTime);
    return start < bookedEnd && end > bookedStart;
  });
}
function cabinStatusLabel(cabinId, date){
  const bookingsToday = cabinRoomBookings.filter(b=>b.cabinId===cabinId && b.date===date && b.status!=='rejected').sort((a,b)=>a.startTime.localeCompare(b.startTime));
  if(!bookingsToday.length) return 'Book';
  const next = bookingsToday[0];
  const label = roomBookingRangeLabel(next.startTime, next.endTime);
  return next.status==='pending_approval' ? `Pending ${label}` : `Booked ${label}`;
}
function findCabinLocation(cabinId){
  for(const building of Object.keys(cabinRooms)){
    for(const floor of Object.keys(cabinRooms[building])){
      const found = (cabinRooms[building][floor]||[]).find(c=>c.id===cabinId);
      if(found) return {cabin:found, building, floor};
    }
  }
  return null;
}

/* =========================================================
   AUTO-RELEASE / NO-SHOW ENGINE
========================================================= */
function runAutoRelease(){
  let released = 0;
  bookings.forEach(b=>{
    if(b.status==='held' && simTime > b.checkinDeadline){
      b.status = 'no_show';
      released++;
      const emp = empById(b.empId);
      if(emp){
        emp.noShow++;
        if(emp.noShow>=3 && !emp.restrictedUntil){
          const restrict = new Date(simTime); restrict.setDate(restrict.getDate()+7);
          emp.restrictedUntil = isoDate(restrict);
          if(emp.id===personaObj().id){
            toast(`${emp.name}: 3 no-shows in a rolling period — booking access suspended for 1 week.`,'bad');
          }
        }
      }
    }
  });
  if(released>0 && document.getElementById('view-booking').classList.contains('active')) renderBooking();
  return released;
}
/* =========================================================
   SHIFT-ENDING-SOON POPUP
   When a checked-in employee's booked shift/half-day window is
   ~15 minutes from ending, prompt them to continue (extend 1
   hour, only if the seat stays free) or leave the seat now.
========================================================= */
function checkShiftEndingSoon(){
  if(document.getElementById('shiftEndBackdrop').classList.contains('active')) return; // one prompt at a time
  const persona = personaObj();
  if(!persona) return;
  const candidate = bookings.find(b=>
    b.empId===persona.id && b.status==='checked_in' && !b.notifiedEndingSoon &&
    (b.shiftEndAt - simTime) <= 15*60000
  );
  if(candidate){
    candidate.notifiedEndingSoon = true;
    openShiftEndModal(candidate.id);
  }
}
function canExtendBooking(b){
  const seat = seats.find(s=>s.id===b.seatId);
  if(!seat || seat.oos || seat.reserved) return false;
  const newEnd = new Date(b.shiftEndAt.getTime() + 60*60000);
  const newEndDateStr = isoDate(newEnd);
  if(newEndDateStr !== b.date){
    // the extra hour spills into the next calendar date — only allow if the seat is free then
    if(seatStatusOnDate(seat, newEndDateStr) !== 'available') return false;
  }
  return true;
}
function openShiftEndModal(bookingId){
  const b = bookings.find(x=>x.id===bookingId);
  if(!b) return;
  const minsLeft = Math.max(0, Math.round((b.shiftEndAt - simTime)/60000));
  const extendable = canExtendBooking(b);
  document.getElementById('shiftEndBody').innerHTML = `
    <div style="text-align:center;padding:4px 0 6px;">
      <div style="font-size:36px;line-height:1;margin-bottom:8px;">⏰</div>
      <h3 style="margin-bottom:2px;">Your shift is ending soon</h3>
      <p class="sub">Seat ${b.seatId} · ${minsLeft>0?`${minsLeft} minute${minsLeft===1?'':'s'} left`:'time is up'} · ends at ${b.endTime}</p>
    </div>
    <div class="banner ${extendable?'banner-info':'banner-warn'}">${extendable ? 'Continue at this seat for 1 more hour, or leave now so it can be released.' : 'This seat is needed after your shift — please wrap up and leave so it can be released.'}</div>
    <div class="modal-actions">
      <button class="btn btn-ghost" style="flex:1;" onclick="leaveSeatNow('${b.id}')">Leave seat</button>
      <button class="btn btn-primary" style="flex:1;" onclick="extendBooking('${b.id}')" ${extendable?'':'disabled'}>Continue (+1 hour)</button>
    </div>
  `;
  document.getElementById('shiftEndBackdrop').classList.add('active');
}
function closeShiftEndModal(){
  document.getElementById('shiftEndBackdrop').classList.remove('active');
}
function extendBooking(bookingId){
  const b = bookings.find(x=>x.id===bookingId);
  if(!b) return;
  if(!canExtendBooking(b)){
    toast(`Seat ${b.seatId} isn't free for another hour — please wrap up soon.`,'bad');
    closeShiftEndModal();
    return;
  }
  b.endTime = addHoursToTime(b.endTime, 1);
  b.shiftEndAt = new Date(b.shiftEndAt.getTime() + 60*60000);
  b.extended = true;
  b.notifiedEndingSoon = false; // allow a fresh "ending soon" prompt near the new end time
  toast(`Extended — ${b.seatId} held until ${b.endTime}.`,'good');
  closeShiftEndModal();
  renderMyBookings(); refreshCurrentView();
}
function leaveSeatNow(bookingId){
  const b = bookings.find(x=>x.id===bookingId);
  if(b){
    b.status = 'cancelled';
    toast(`Checked out of ${b.seatId} — seat released.`,'info');
  }
  closeShiftEndModal();
  renderMyBookings(); refreshCurrentView();
}
function advanceClock(){
  simTime = new Date(simTime.getTime() + 30*60000);
  updateClock();
  const released = runAutoRelease();
  if(released>0){
    toast(`Auto-release: ${released} seat(s) reclaimed after missed check-in window.`,'warn');
  }
  checkShiftEndingSoon();
  refreshCurrentView();
}
function updateClock(){
  document.getElementById('clockDisplay').textContent =
    simTime.toLocaleDateString('en-IN',{day:'2-digit',month:'short'}) + '  ' + fmtTime(simTime);
  renderOfficeEntryBox();
}

/* =========================================================
   OFFICE ENTRY (ID CARD / BIOMETRIC) — DEMO
   Check-in is only available once the logged-in employee has
   "scanned in" at the office for today.
========================================================= */
function isScannedInToday(empId){
  return !!(officeCheckins[empId] && officeCheckins[empId][todayStr]);
}
function officeScanTime(empId){
  return (officeCheckins[empId] && officeCheckins[empId][todayStr]) || null;
}
function scanOfficeEntry(){
  const emp = personaObj();
  if(!officeCheckins[emp.id]) officeCheckins[emp.id] = {};
  officeCheckins[emp.id][todayStr] = fmtTime(simTime);
  toast(`ID card / biometric scanned at ${fmtTime(simTime)} — check-in is now available.`, 'good');

  // Auto check-in: scanning the ID card / biometric immediately checks the
  // employee into any of today's held bookings (if still within the
  // check-in window), which flips the seat to "seated"/occupied and the
  // booking status to "checked-in" — no separate manual check-in click needed.
  const todays = bookings.filter(b=>b.empId===emp.id && b.date===todayStr && b.status==='held');
  todays.forEach(b=>{
    if(simTime > b.checkinDeadline){
      toast(`Check-in window already closed for ${b.seatId} — couldn't auto check-in.`,'warn');
      return;
    }
    b.status='checked_in';
    toast(`Auto checked in to ${b.seatId}. Enjoy the day!`,'good');
  });

  renderOfficeEntryBox();
  renderMyBookings();
  refreshCurrentView();
}
function resetOfficeEntry(){
  const emp = personaObj();
  if(officeCheckins[emp.id]) delete officeCheckins[emp.id][todayStr];
  toast('Office entry reset for demo purposes.', 'info');
  renderOfficeEntryBox();
  refreshCurrentView();
}
function renderOfficeEntryBox(){
  const box = document.getElementById('officeEntryBox');
  if(!box) return;
  const emp = personaObj();
  const scannedAt = officeScanTime(emp.id);
  box.innerHTML = `
    Office entry (demo)
    ${scannedAt
      ? `<div class="office-entry-status in">✅ Scanned in — ${emp.name}<span class="office-entry-time">at ${scannedAt} today</span></div>
         <button class="rail-clock-btn" onclick="resetOfficeEntry()">↺ Reset (simulate not scanned)</button>`
      : `<div class="office-entry-status out">⛔ Not scanned in yet</div>
         <button class="rail-clock-btn" onclick="scanOfficeEntry()">🪪 Simulate ID card / biometric scan</button>`
    }
  `;
}

function computeUtilization(date){
  const bookable = seats.filter(s=>!s.reserved && !s.oos);
  const occ = bookable.filter(s=>{
    const st = seatStatusOnDate(s,date);
    return st==='held'||st==='occupied';
  }).length;
  return Math.round((occ/bookable.length)*100);
}
function computeNoShowRate(){
  const total = bookings.length;
  const ns = bookings.filter(b=>b.status==='no_show').length;
  return total? Math.round((ns/total)*1000)/10 : 0;
}
function computeBtoO(){
  const relevant = bookings.filter(b=>['checked_in','no_show'].includes(b.status));
  const checked = bookings.filter(b=>b.status==='checked_in').length;
  return relevant.length? Math.round((checked/relevant.length)*1000)/10 : 100;
}

let activeStatusFilters = ['all']; // multi-select: subset of 'available' | 'booked' | 'blocked' | 'maintenance', or ['all']
function toggleStatusFilter(key){
  if(key==='all'){
    activeStatusFilters = ['all'];
  } else {
    activeStatusFilters = activeStatusFilters.filter(k=>k!=='all');
    const idx = activeStatusFilters.indexOf(key);
    if(idx>=0) activeStatusFilters.splice(idx,1);
    else activeStatusFilters.push(key);
    if(!activeStatusFilters.length) activeStatusFilters = ['all'];
  }
  renderBooking();
}

function computeDatesForMode(mode, baseDate, fromDate, toDate, monthStr){
  if(mode==='full') return [baseDate];
  if(mode==='multi'){
    if(!fromDate || !toDate || toDate < fromDate) return [];
    const dates=[];
    let d = new Date(fromDate+'T00:00:00');
    const end = new Date(toDate+'T00:00:00');
    while(d<=end){
      const day = d.getDay();
      if(day!==0 && day!==6) dates.push(isoDate(d));
      d.setDate(d.getDate()+1);
    }
    return dates;
  }
  if(mode==='month'){
    if(!monthStr) return [];
    const [y,m] = monthStr.split('-').map(Number);
    const dates=[];
    const last = new Date(y, m, 0).getDate();
    for(let day=1; day<=last; day++){
      const d = new Date(y, m-1, day);
      const wd = d.getDay();
      if(wd!==0 && wd!==6) dates.push(isoDate(d));
    }
    return dates;
  }
  return [baseDate];
}

function getFloorCabins(building, floor){
  return (cabinRooms[building] && cabinRooms[building][floor]) ? cabinRooms[building][floor] : [];

function findCabinById(cabinId){
  for(const building of Object.keys(cabinRooms)){
    for(const floor of Object.keys(cabinRooms[building])){
      const found = (cabinRooms[building][floor]||[]).find(c=>c.id===cabinId);
      if(found) return {...found, building, floor};
    }
  }
  return null;
}

function seatMatchesFilter(seat, bookingDate){
  if(activeStatusFilters.includes('all')) return true;
  const hrStatus = seat.hrStatus || 'available';
  return activeStatusFilters.some(key=>{
    if(key === 'blocked') return hrStatus === 'blocked';
    if(key === 'maintenance') return hrStatus === 'maintenance';
    if(hrStatus !== 'available') return false; // blocked/maintenance won't match available/booked
    const dynStatus = seatStatusOnDate(seat, bookingDate);
    if(key === 'available') return dynStatus === 'available';
    if(key === 'booked') return dynStatus === 'held' || dynStatus === 'occupied';
    return false;
  });
}

function buildFloorLayoutHtml(floorSeats, emp, restricted, myBookingToday, maxTeamCapacity, bookingDate){
  const cabins = getFloorCabins(selectedBuilding, selectedFloorName);
  const leftCabins = cabins.filter(c=>c.side==='left');
  const rightCabins = cabins.filter(c=>c.side==='right');

  // Group normal seats by zone rows
  const zones = [...new Set(floorSeats.map(s=>s.zone))];

  function renderCabin(cabin){
    const isBlocked = cabin.hrStatus === 'blocked';
    const canBook = !isBlocked && canBookSeatType('cabin');
    const matchesFilter = activeStatusFilters.includes('all') ||
      (activeStatusFilters.includes('blocked') && isBlocked) ||
      (activeStatusFilters.includes('available') && !isBlocked);
    const dimmed = !matchesFilter ? 'cabin-dimmed' : '';
    const blockedClass = isBlocked ? 'cabin-blocked' : '';
    const clickAttr = canBook && matchesFilter ? `onclick="openCabinBookModal('${cabin.id}','${cabin.name.replace(/'/g,"\\'")}')"` : '';
    const statusText = isBlocked ? 'Blocked' : canBook ? cabinStatusLabel(cabin.id, selectedDate) : 'Cabin';
    return `
      <div class="cabin-room ${blockedClass} ${dimmed}" ${clickAttr} title="${cabin.name}${isBlocked?' (Blocked)':''}">
        <div class="cabin-name">${cabin.name}</div>
        <div class="cabin-status">${statusText}</div>
      </div>`;
  }

  function renderSeat(s){
    const hrStatus = s.hrStatus || 'available';
    if(hrStatus === 'blocked'){
      const matchesFilter = activeStatusFilters.includes('all') || activeStatusFilters.includes('blocked');
      return `<div class="seat seat-oos ${matchesFilter?'':'seat-dimmed'}" data-seat-id="${s.id}" title="${s.id} - Blocked by HR"><div class="s-id">${s.id.split('-').pop()}</div><div class="s-status">Blocked</div></div>`;
    }
    if(hrStatus === 'maintenance'){
      const matchesFilter = activeStatusFilters.includes('all') || activeStatusFilters.includes('maintenance');
      return `<div class="seat seat-oos ${matchesFilter?'':'seat-dimmed'}" data-seat-id="${s.id}" title="${s.id} - Under maintenance"><div class="s-id">${s.id.split('-').pop()}</div><div class="s-status">Maint.</div></div>`;
    }
    const status = seatStatusOnDate(s, bookingDate);
    const booking = bookings.find(b=>b.seatId===s.id && b.date===bookingDate && (b.status==='held'||b.status==='checked_in'));
    const selectedForTeam = selectedTeamSeats.includes(s.id);
    const mine = booking && booking.empId===emp.id;
    const clickable = status==='available' && !restricted && (!myBookingToday || bookingTargetEmpId==='team') && (bookingTargetEmpId!=='team' || maxTeamCapacity>0);
    const label = {available:'Open', held:'Booked', occupied:'Seated', vip:'Reserved', oos:'Out of use'}[status];
    const matchesFilter = seatMatchesFilter(s, bookingDate);
    const dimClass = matchesFilter ? '' : 'seat-dimmed';
    return `<div class="seat seat-${status} ${mine?'seat-mine':''} ${selectedForTeam?'seat-selected':''} ${dimClass}"
              data-seat-id="${s.id}"
              title="${s.id} - ${label}"
              ${clickable && matchesFilter ?`onclick="${bookingTargetEmpId==='team' ? `toggleTeamSeat('${s.id}')` : `openBookModal('${s.id}')`}"`:''}>
              <div class="s-id">${s.id.split('-').pop()}</div><div class="s-status">${selectedForTeam?'Pick':label}</div>
            </div>`;
  }

  const leftCabinHtml = leftCabins.map(renderCabin).join('');
  const rightCabinHtml = rightCabins.map(renderCabin).join('');

  const zoneRows = zones.map(zoneName=>{
    const zSeats = floorSeats.filter(s=>s.zone===zoneName);
    const avail = zSeats.filter(s=>s.hrStatus==='available' && seatStatusOnDate(s,bookingDate)==='available').length;
    const total = zSeats.length;
    return `
      <div class="zone" style="margin-bottom:12px;">
        <div class="zone-head">
          <div class="zone-title">${zoneName}</div>
          <div class="zone-meta">${avail}/${total} open</div>
        </div>
        <div class="seat-row">${zSeats.map(renderSeat).join('')}</div>
      </div>`;
  }).join('');

  return `
    <div class="floor-plan-card">
      <div class="floor-layout-container">
        <div class="floor-layout-left">${leftCabinHtml || '<div class="no-cabins"></div>'}</div>
        <div class="floor-walkway"></div>
        <div class="floor-layout-center">${zoneRows}</div>
        <div class="floor-walkway"></div>
        <div class="floor-layout-right">${rightCabinHtml || '<div class="no-cabins"></div>'}</div>
      </div>
    </div>
  `;
}

function confirmCabinBooking(){
  const cabin = findCabinById(pendingCabinRoom);
  if(!cabin) return;
  const endTime = meetingRoomEndTime(cabinRoomStartTime, cabinRoomDuration);
  if(cabinBookingOverlaps(cabin.id, selectedDate, cabinRoomStartTime, endTime)){
    toast('That time overlaps with another cabin booking.','bad');
    renderCabinBookModal();
    return;

function computeDurationDates(){
  if(durationMode==='single') return [selectedDate];
  if(durationMode==='multi'){
    if(!multiFrom || !multiTo || multiTo < multiFrom || !multiWeekdays.length) return [];
    const interval = Math.max(1, multiRepeatInterval|0);
    const start = new Date(multiFrom+'T00:00:00');
    const end = new Date(multiTo+'T00:00:00');
    // Monday of the week containing the start date (so "week 0" always includes the start date)
    const startDow = start.getDay(); // 0=Sun..6=Sat
    const mondayOffset = startDow===0 ? 6 : startDow-1;
    const anchorMonday = new Date(start); anchorMonday.setDate(anchorMonday.getDate()-mondayOffset);
    const dates=[];
    let d = new Date(start);
    while(d<=end){
      if(multiWeekdays.includes(d.getDay())){
        const diffDays = Math.round((d - anchorMonday)/86400000);
        const weekIndex = Math.floor(diffDays/7);
        if(weekIndex>=0 && weekIndex % interval === 0) dates.push(isoDate(d));
      }
      d.setDate(d.getDate()+1);
    }
    return dates;
  }
  if(durationMode==='month'){
    if(!monthValue) return [];
    const [y,m] = monthValue.split('-').map(Number);
    const dates=[];
    const last = new Date(y, m, 0).getDate();
    for(let day=1; day<=last; day++){
      const d = new Date(y, m-1, day);
      const wd = d.getDay();
      if(wd!==0 && wd!==6) dates.push(isoDate(d));
    }
    return dates;
  }
  return [selectedDate];
}

function confirmBooking(){
  if(collisionTriggered){ toast('Please pick a different seat.','bad'); return; }
  const seat = seats.find(s=>s.id===pendingSeat);
  const target = bookingTargetObj();
  const targetDates = effectiveDurationDates();
  if(!targetDates.length){ toast(durationMode==='multi' ? 'Select at least one day of the week to book.' : 'Pick a valid date range.','bad'); return; }
  const dateStatuses = targetDates.map(dt=>({date:dt, status:durationDateStatus(seat, target.id, dt)}));
  const availableDates = dateStatuses.filter(x=>x.status==='available').map(x=>x.date);
  const skipped = dateStatuses.filter(x=>x.status!=='available').map(x=>x.date);
  if(!availableDates.length){
    collisionTriggered = true; renderModal();
    toast('No open days in this range — please pick another seat or date range.','bad');
    return;
  }
  const slot = SHIFTS[bookingShift];
  const effStart = slot.start;
  const effEnd = dayLengthMode==='half' ? addHoursToTime(slot.start, halfDayHours) : slot.end;
  const lengthLabel = dayLengthMode==='half' ? `Half day (${halfDayHours}h)` : 'Full day';
  const modeLabel = durationMode==='multi' ? `Multi-Day booking · ${slot.label} shift · ${lengthLabel}`
    : durationMode==='month' ? `Month booking · ${slot.label} shift · ${lengthLabel}`
    : `${slot.label} shift · ${lengthLabel}`;
  const groupId = 'GRP-'+(groupSeq++);
  availableDates.forEach(dt=>mkBooking(pendingSeat, target.id, dt, 'held', {
    groupId, slotLabel: modeLabel, startTime: effStart, endTime: effEnd
  }));
  lastConfirmedBooking = {
    seatId: pendingSeat,
    workspace: seat.zone,
    company: ORG_COMPANY,
    location: selectedLocation || LOCATIONS[0],
    building: selectedBuilding || seat.building,
    floor: selectedFloorName || seat.floor,
    targetName: target.name,
    dates: availableDates,
    requestedCount: targetDates.length,
    skipped,
    slotLabel: modeLabel,
    startTime: effStart,
    endTime: effEnd
  };
  bookingConfirmed = true;
  renderModal();
  renderBooking(); renderMyBookings();
}

function renderMyBookings(){

  const mine = bookings.filter(b=>b.empId===p.id).sort((a,b)=>a.date.localeCompare(b.date));

    const pillClass = {held:'pill-held',checked_in:'pill-checked',cancelled:'pill-cancelled',no_show:'pill-noshow'}[b.status];
    const pillLabel = {held:'Booked',checked_in:'Checked in',cancelled:'Cancelled',no_show:'No-show'}[b.status];

          actions += `<span style="color:var(--amber);font-size:11.5px;">🪪 Scan ID card / biometric at office entry to check in</span> `;
        }
      } else if(isToday){
        actions += `<span style="color:var(--red);font-size:11.5px;">Window closed</span> `;
      } else {
        actions += `<span style="color:var(--ink-faint);font-size:11.5px;">Check-in opens on the day</span> `;

function checkInBooking(id){
  const b = bookings.find(x=>x.id===id);
  if(simTime > b.checkinDeadline){ toast('Check-in window has closed for this booking.','bad'); return; }

  b.status='checked_in';
  toast(`Checked in to ${b.seatId}. Enjoy the day!`,'good');
  renderMyBookings(); renderBooking();
}
function cancelBooking(id){

function cancelBookingSeries(groupId){

// Builds a basic default layout for a freshly created floor: cabins on the left
// and right (same pattern as existing floors), plain seat zones in the middle.
// Every generated seat/cabin starts Blocked so HR must consciously open the floor.
function generateDefaultFloorLayout(building, floorName){
  const bldgIdx = BUILDINGS.indexOf(building) + 1;
  const floorTag = floorName.replace(/[^A-Za-z0-9]/g,'').toUpperCase().slice(0,6) || 'FLR';
  const prefixBase = `B${bldgIdx}${floorTag}`;
  const zoneConfigs = [
    {prefix:`${prefixBase}-GENA`, count:ROW_COUNT, zone:'General Pod A', dept:'General'},
    {prefix:`${prefixBase}-GENB`, count:ROW_COUNT, zone:'General Pod B', dept:'General'}
  ];
  const cabins = [
    {id:`CAB-${prefixBase}-1`, name:'Meeting Room 1', side:'left', seatType:'cabin', hrStatus:'blocked'},
    {id:`CAB-${prefixBase}-2`, name:'Meeting Room 2', side:'left', seatType:'cabin', hrStatus:'blocked'},
    {id:`CAB-${prefixBase}-3`, name:'Meeting Room 3', side:'right', seatType:'cabin', hrStatus:'blocked'},
    {id:`CAB-${prefixBase}-4`, name:'Meeting Room 4', side:'right', seatType:'cabin', hrStatus:'blocked'}
  ];
  return {zoneConfigs, cabins};
}
function hrAddFloor(){
  const bldg = document.getElementById('hrFloorBldg').value;
  const fname = document.getElementById('hrFloorName').value.trim();
  if(!fname){ toast('Floor name required.','bad'); return; }
  if(!FLOOR_LAYOUTS[bldg]) FLOOR_LAYOUTS[bldg] = {};
  if(FLOOR_LAYOUTS[bldg][fname]){ toast('Floor already exists in this building.','bad'); return; }

  const {zoneConfigs, cabins} = generateDefaultFloorLayout(bldg, fname);
  FLOOR_LAYOUTS[bldg][fname] = zoneConfigs;

  // Generate the actual seat records for the new floor and drop them straight
  // into the shared `seats` array so they show up in Seat Management immediately —
  // no manual sync step required. All generated seats start Blocked.
  zoneConfigs.forEach(cfg=>{
    workspaceMeta[cfg.zone] = `${cfg.dept} · ${bldg} · ${fname}`;
    const newSeats = zoneSeats(cfg.prefix, cfg.count, cfg.zone, cfg.dept, bldg, fname, LOCATIONS[0], 0, [], 'normal', []);
    newSeats.forEach(s=>{ s.hrStatus = 'blocked'; });
    seats.push(...newSeats);
  });

  // Generate the default cabins (left/right), also Blocked until HR opens them up.
  if(!cabinRooms[bldg]) cabinRooms[bldg] = {};
  cabinRooms[bldg][fname] = cabins;

  toast(`${fname} added to ${bldg} with a default layout. All seats start Blocked — update statuses in Seat Management when ready.`, 'good');
  renderHRBuildings();
}
function hrDeleteFloor(bldg, floor){
  if(!confirm(`Delete ${floor} in ${bldg}?`)) return;
  if(FLOOR_LAYOUTS[bldg]) delete FLOOR_LAYOUTS[bldg][floor];
  seats = seats.filter(s=>!(s.building===bldg && s.floor===floor));
  if(cabinRooms[bldg]) delete cabinRooms[bldg][floor];
  toast(`${floor} removed from ${bldg}.`, 'info');
  renderHRBuildings();
}

function hrSetSeatStatus(seatId, status){
  const seat = seats.find(s=>s.id===seatId);
  if(!seat) return;
  const old = seat.hrStatus;
  seat.hrStatus = status;
  if(status==='blocked' || status==='maintenance'){
    // auto-cancel active bookings for that seat
    const today = todayStr;
    const affected = bookings.filter(b=>b.seatId===seatId && b.date>=today && (b.status==='held'||b.status==='checked_in'));
    affected.forEach(b=>{ b.status='cancelled'; });
    if(affected.length) toast(`${affected.length} booking(s) for ${seatId} auto-cancelled.`,'warn');
  }
  toast(`${seatId} status changed to ${status}.`, 'info');
  renderHRSeats();
}

function hrApproveCabinRequest(id){
  const b = cabinRoomBookings.find(x=>x.id===id);
  if(!b) return;
  b.status = 'confirmed';
  const loc = findCabinLocation(b.cabinId);
  toast(`${loc ? loc.cabin.name : b.cabinId} request approved. Employee has been notified.`, 'good');
  renderHRBookings(); renderMyBookings(); renderBooking();
}
function hrRejectCabinRequest(id){
  const b = cabinRoomBookings.find(x=>x.id===id);
  if(!b) return;
  const reason = prompt('Reason for rejecting this request (optional):') || '';
  b.status = 'rejected';
  b.rejectionReason = reason.trim() || null;
  const loc = findCabinLocation(b.cabinId);
  toast(`${loc ? loc.cabin.name : b.cabinId} request rejected. The room remains available and the employee has been notified.`, 'info');
  renderHRBookings(); renderMyBookings(); renderBooking();
}
function hrForceCancel(id){
  const b = bookings.find(x=>x.id===id);
  if(!b) return;
  b.status='cancelled';
  toast(`Booking ${b.id} cancelled by HR.`,'info');
  renderHRBookings(); renderBooking(); renderMyBookings();
}
function forceCancel(id){ hrForceCancel(id); }

function onGlobalSearch(value){
  const term = value.trim().toLowerCase();
  const resultsEl = document.getElementById('globalSearchResults');
  const clearBtn = document.getElementById('globalSearchClearBtn');
  clearBtn.style.display = term ? 'block' : 'none';
  if(!term){ resultsEl.classList.remove('active'); resultsEl.innerHTML=''; return; }

  // HR search only ever routes to Employee Management, Building Management, or Seat Management.
  const empMatches = employees.filter(e=>e.id.toLowerCase().includes(term) || e.name.toLowerCase().includes(term)).slice(0,5);
  const buildingMatches = BUILDINGS.filter(b=>b.toLowerCase().includes(term)).slice(0,5);
  const seatMatches = seats.filter(s=>s.id.toLowerCase().includes(term)).slice(0,6);

  let html = '';
  if(empMatches.length){
    html += `<div class="gsr-group-label">Employees</div>` + empMatches.map(e=>`
      <div class="gsr-item" onclick="goToSearchResult('employee','${e.id}')">
        <div><div class="gsr-title">${e.name}</div><div class="gsr-sub">${e.id} · ${e.assignedBuilding||'—'} · ${e.assignedFloor||'—'}</div></div>
        <span>›</span>
      </div>`).join('');
  }
  if(buildingMatches.length){
    html += `<div class="gsr-group-label">Buildings</div>` + buildingMatches.map(b=>`
      <div class="gsr-item" onclick="goToSearchResult('building','${b}')">
        <div><div class="gsr-title">${b}</div><div class="gsr-sub">Open in Building Management</div></div>
        <span>›</span>
      </div>`).join('');
  }
  if(seatMatches.length){
    html += `<div class="gsr-group-label">Seats</div>` + seatMatches.map(s=>`
      <div class="gsr-item" onclick="goToSearchResult('seat','${s.id}')">
        <div><div class="gsr-title">${s.id}</div><div class="gsr-sub">${s.zone} · ${s.building} · ${s.floor}</div></div>
        <span>›</span>
      </div>`).join('');
  }
  if(!html){
    html = `<div class="gsr-empty">No matches for "${value}"</div>`;
  }
  resultsEl.innerHTML = html;
  resultsEl.classList.add('active');
}
function clearGlobalSearch(){
  const input = document.getElementById('globalSearchInput');
  input.value = '';
  onGlobalSearch('');
  input.focus();
}
function closeGlobalSearchResults(){
  const resultsEl = document.getElementById('globalSearchResults');
  if(resultsEl) resultsEl.classList.remove('active');
}
function goToSearchResult(type, id){
  closeGlobalSearchResults();
  const input = document.getElementById('globalSearchInput');
  if(input) input.value = '';
  const clearBtn = document.getElementById('globalSearchClearBtn');
  if(clearBtn) clearBtn.style.display = 'none';

  if(type==='employee'){
    const emp = empById(id);
    if(!emp) return;
    if(currentRole==='hr'){
      showView('hr-employees');
      setTimeout(()=>highlightHREmployeeRow(emp.id), 60);
    } else {
      toast(`${emp.name} (${emp.id}) is assigned to ${emp.assignedBuilding} · ${emp.assignedFloor}.`, 'info');
    }
  } else if(type==='building'){
    // Building search always opens Building Management — never the booking/floor-plan view.
    showView('hr-buildings');
    setTimeout(()=>highlightHRBuildingRow(id), 60);
  } else if(type==='seat'){
    // Seat search always opens Seat Management for administrative control — never Book a Seat.
    const seat = seats.find(s=>s.id===id);
    if(!seat) return;
    showView('hr-seats');
    setTimeout(()=>highlightHRSeatRow(seat.id), 60);
  }
}

  });
}
